# P3 - HYPOTHESIS TESTING
HYPOTHESIS prac 3
import pandas as pd
import numpy as np
from scipy.stats import ttest_ind
df = pd.read_csv("students_data.csv")
print(df)
ai_scores = df[df["Study_Method"] == "AI_Module"]["Final_Score"]
traditional_scores = df[df["Study_Method"] == "Traditional"]["Final_Score"]
mean_ai = np.mean(ai_scores)
mean_traditional = np.mean(traditional_scores)
print("AI Module Mean Score:", mean_ai)
print("Traditional Mean Score:", mean_traditional)
print("Mean Difference:", mean_ai - mean_traditional)
t_stat, p_value_two_tailed = ttest_ind(
    ai_scores,
    traditional_scores,
)
if t_stat > 0:
    p_value_one_tailed = p_value_two_tailed / 2
else:
    p_value_one_tailed = 1 - (p_value_two_tailed / 2)
print("\nHypothesis Test")
print("t-statistic:", t_stat)
print("One-tailed p-value:", p_value_one_tailed)
alpha = 0.05
if p_value_one_tailed < alpha:
    print("Decision: Reject H0")
    print(
        "Conclusion: Students using the AI Module score significantly higher than students using traditional study materials."
    )
else:
    print("\nDecision: Fail to reject H0")
    print("Conclusion: There is insufficient evidence that the AI Module produces higher scores."
    )

# P4 - LINEAR REGRESSION
Linear Regression prac 4
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score
df = pd.read_csv("house_data.csv")
print(df)
X = df[
    [
        "Square_Feet",
        "Bedroom_Count",
        "Property_Age",
        "Distance_To_City_Center"
    ]
]
y = df["Listing_Price"]
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42
)
model = LinearRegression()
model.fit(X_train, y_train)
print("Coefficient: ",model.coef_[0])
print("Intercept:", model.intercept_)
y_pred = model.predict(X_test)
print("Actual Prices: ")
print(y_test.values)
print("\n Predicted Prices: ")
print(y_pred)
results = pd.DataFrame({
    "Actual Price": y_test.values,
    "Predicted Price": y_pred
})
print("\nActual vs Predicted Prices:")
print(results.round(2))
mse = mean_squared_error(y_test, y_pred)
print("MSE: ",mse)
rmse = np.sqrt(mse)
print("RMSE: ",rmse)
r2 = r2_score(y_test, y_pred)
print("R2 score: ",r2)
plt.scatter(y_test, y_pred)
plt.xlabel("Actual Price (₹ lakh)")
plt.ylabel("Predicted Price (₹ lakh)")
plt.title("Actual vs Predicted House Prices")
plt.show()

    ],

        25, 28, 23, 30, 27,
        15, 18, 16, 20, 17,

        1800, 2000, 1750, 1900, 2100,
        800, 820, 1650, 1700, 1100
    ]

    ["Annual_Income_k", "Annual_Visits"]
]

inertia = []

for k in K_range:
    kmeans = KMeans(
        n_clusters=k,
        random_state=42,
        n_init=10
    )
    kmeans.fit(X_scaled)
    inertia.append(kmeans.inertia_)

plt.plot(
    K_range,
    inertia,
    marker="o"
)
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method - Customer Segmentation")
plt.xticks(K_range)
plt.show()

    n_clusters=4,
    random_state=42,
    n_init=10
)

)

    centers,
    columns=[
        "Annual_Income_k",
        "Annual_Visits"
    ]
)

)

plt.scatter(
    df["Annual_Income_k"],
    df["Annual_Visits"],
    c=df["Cluster"],
    s=100
)
plt.scatter(
    cluster_centers["Annual_Income_k"],
    cluster_centers["Annual_Visits"],
    marker="X",
    s=250,
    label="Cluster Centers"
)

)

# P7 - MAPREDUCE
MapReduce prac 7
from pyspark.sql import SparkSession
spark = SparkSession.builder.appName("Practical_1").getOrCreate()
sales = [
    ("P101", "2", "500"),
    ("P102", "3", "200"),
    ("P101", "4", "500"),
    ("P103", "5", "100"),
    ("P102", "2", "200"),
    ("P103", "3", "100")
]
rdd = spark.sparkContext.parallelize(sales)
pairs = rdd.map(
    lambda x: (x[0], int(x[1]) * float(x[2]))
)
print("Product Transaction Values:")
print(pairs.collect())
result = pairs.reduceByKey(lambda a, b: a + b)
print("\nTotal Revenue Per Product:")
for product, revenue in sorted(result.collect()):
    print(product, ":", revenue)
spark.stop()

    (105, 2, "2026-03-01", 30000),
    (106, 4, "2026-03-05", 60000),
    (107, 5, "2026-03-12", 20000)
]
order_cols = [
    "OrderID", "CustomerID", "OrderDate", "TotalAmount"
]
df_orders = spark.createDataFrame(orders, order_cols)
customers = [
    (1, "Amit", "India"),
    (2, "Neha", "India"),
    (3, "Rahul", "USA"),
    (4, "Priya", "India"),
    (5, "Karan", "India")
]
customer_cols = ["CustomerID", "CustomerName", "Country"]
df_customers = spark.createDataFrame(
    customers, customer_cols
)
print("Orders Data:")
df_orders.show()
print("Customers Data:")
df_customers.show()
df_orders.createOrReplaceTempView("orders")
df_customers.createOrReplaceTempView("customers")
query = """
SELECT
    c.CustomerID,
    c.CustomerName,
    c.Country,
    SUM(o.TotalAmount) AS Total_Spending
FROM orders o
INNER JOIN customers c
    ON o.CustomerID = c.CustomerID
WHERE c.Country = 'India'
GROUP BY
    c.CustomerID,
    c.CustomerName,
    c.Country
HAVING SUM(o.TotalAmount) > 50000
ORDER BY Total_Spending DESC
"""
result = spark.sql(query)
print("Indian Customers Spending More Than 50000:")
result.show()
spark.stop()

# P10 - AVERAGE USING RDD
prac 10 average
from pyspark.sql import SparkSession
spark = SparkSession.builder.appName("Practical_4").getOrCreate()
scores = [
    ("Alice", 85),
    ("Bob", 70),
    ("Alice", 90),
    ("Bob", 80),
    ("Charlie", 95)
]
rdd = spark.sparkContext.parallelize(scores)
pairs = rdd.map(
    lambda x: (x[0], (x[1], 1))
)
result = pairs.reduceByKey(
    lambda a, b: (a[0] + b[0], a[1] + b[1])
)
average = result.map(
    lambda x: (x[0], x[1][0] / x[1][1])
)
print("Average Score of Each Student:")
for name, avg_score in sorted(average.collect()):
    print(name, ":", avg_score)
spark.stop()
