# PART 2: FUZZY MEMBERSHIP

print("FUZZY MEMBERSHIP")

def hot_membership(temp):

    if temp <= 25:
        return 0

    elif temp >= 40:
        return 1

    else:
        return (temp - 25) / (40 - 25)


fuzzy_output = hot_membership(temperature)

print("Temperature =", temperature, "°C")
print("Fuzzy Membership in HOT =", round(fuzzy_output, 2))

# PRACTICAL 12: JOINT AND CONDITIONAL PROBABILITY
# ============================================================

print("Vandana Narang 26K4RIT005")

def conditional():

    pass_stats = 0.15
    pass_codingWStats = 0.60
    pass_codingWOStats = 0.40

    # Probability of passing both Statistics and Coding

    prob_both = pass_stats * pass_codingWStats

    print("The probability that applicant passes both is",
          round(prob_both, 3))

    # Probability of passing Coding

    prob_coding = prob_both + ((1 - pass_stats) * pass_codingWOStats)

    print("The probability that he/she passes Coding is",
          round(prob_coding, 3))

    # Conditional probability: P(Statistics | Coding)

    stats_given_coding = prob_both / prob_coding

    print("Conditional probability is",
          round(stats_given_coding, 3))

    print("Vandana Narang 26K4RIT005")


conditional()

def bayes_theorem(prior_prob, likelihood, evidence):

    posterior_prob = (likelihood * prior_prob) / evidence

    return posterior_prob
